local function new(class, props, parent)
	local o = Instance.new(class)
	for k,v in pairs(props) do o[k] = v end
	if parent then o.Parent = parent end
	return o
end
local function charOf(plr)
	local c = plr.Character
	if not c then return nil end
	local hum = c:FindFirstChildOfClass("Humanoid")
	local hrp = c:FindFirstChild("HumanoidRootPart")
	if hum and hrp and hum.Health > 0 then return c, hum, hrp end
	return nil
end
local function isTeam(p) return p.Team ~= nil and LP.Team ~= nil and p.Team == LP.Team end
local function keyName(k)
	if typeof(k) == "EnumItem" then return (tostring(k):match("%.(.+)$") or "?") end
	return "?"
end
local function keyMatch(i, k) return i.KeyCode == k or i.UserInputType == k end
local function getPart(c, n) return c:FindFirstChild(n) or c:FindFirstChild("HumanoidRootPart") end

local function visibleTo(part, char)
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Exclude
	rp.FilterDescendantsInstances = {LP.Character}
	local hit = workspace:Raycast(Camera.CFrame.Position, part.Position - Camera.CFrame.Position, rp)
	if not hit then return true end
	return hit.Instance:IsDescendantOf(char)
end

local randAt, randVal = 0, 0.5
local function randFrac()
	local t = os.clock()
	if t - randAt > 0.22 then randAt = t; randVal = math.random() end
	return randVal
end

local function targetPos(char, cfg)
	if cfg.randomPart then
		local head = char:FindFirstChild("Head")
		local hrp = char:FindFirstChild("HumanoidRootPart")
		if head and hrp then
			local base = head.Position:Lerp(hrp.Position - Vector3.new(0, 1.5, 0), randFrac())
			return base + Vector3.new(math.random()*0.5-0.25, math.random()*0.5-0.25, math.random()*0.5-0.25)
		end
	end
	return getPart(char, cfg.part).Position
end

local function findTarget(cfg)
	local best, bestD = nil, math.huge
	local vp = Camera.ViewportSize
	local center = Vector2.new(vp.X/2, vp.Y/2)
	local camPos = Camera.CFrame.Position
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP and not (cfg.team and isTeam(plr)) then
			local char = charOf(plr)
			if char then
				local part = getPart(char, cfg.part)
				local d3 = (camPos - part.Position).Magnitude
				if (not cfg.maxDist or cfg.maxDist <= 0 or d3 <= cfg.maxDist) then
					local pos = targetPos(char, cfg)
					local sp, on = Camera:WorldToViewportPoint(pos)
					if on then
						local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
						if d <= cfg.fov and d < bestD then
							if (not cfg.wall) or visibleTo(part, char) then
								best, bestD = plr, d
							end
						end
					end
				end
			end
		end
	end
	return best
end

--// SILENT AIM
local silentPos, silentAt = nil, 0
local function silentTargetPos()
	if not S.silent.on then return nil end
	local now = os.clock()
	if now - silentAt < 0.015 then return silentPos end
	silentAt = now
	silentPos = nil
	if math.random(1,100) > S.silent.chance then return nil end
	local plr = findTarget(S.silent)
	if not plr then return nil end
	local char = plr.Character
	if not char then return nil end
	silentPos = targetPos(char, S.silent)
	return silentPos
end

					or m == "FindPartOnRayWithWhitelist") and S.silent.hookRay then
					local ray = ...
					if typeof(ray) == "Ray" then
						local t = silentTargetPos()
						if t then return old(self, Ray.new(ray.Origin, (t - ray.Origin).Unit * ray.Direction.Magnitude), select(2, ...)) end
					end
				elseif (m == "ScreenPointToRay" or m == "ViewportPointToRay") and S.silent.hookSPR then
					local ray = old(self, ...)
					local t = silentTargetPos()
					if t and typeof(ray) == "Ray" then
						return Ray.new(ray.Origin, (t - ray.Origin).Unit * ray.Direction.Magnitude)
					end
					return ray
				end
			end
			return old(self, ...)
		end)
		setreadonly(mt, true)
	end)
	if not ok then warn("[VERSACE] silent aim: "..tostring(err)) end
end

local function aimActive() return S.aim.on and ((not S.aim.hold) or aimHeld) end
local function runAim()
	local plr = findTarget(S.aim)
	if not plr then aimLastTarget = nil; return end
	if S.aim.hitchance then
		if plr ~= aimLastTarget then
			aimLastTarget = plr
			aimPass = math.random(1,100) <= S.aim.hitchanceVal
		end
		if not aimPass then return end
	else
		aimLastTarget = plr
	end
	local char = plr.Character
	if not char then return end
	local pos = targetPos(char, S.aim)
	if S.aim.mode == "Mouse" and hasMouseRel then
		local sp = Camera:WorldToViewportPoint(pos)
		local vp = Camera.ViewportSize
		mousemoverel((sp.X - vp.X/2) / S.aim.smooth, (sp.Y - vp.Y/2) / S.aim.smooth)
	else
		Camera.CFrame = Camera.CFrame:Lerp(CFrame.lookAt(Camera.CFrame.Position, pos), math.clamp(1/S.aim.smooth, 0, 1))
	end
end

local function delESP(plr)
	local o = espCache[plr]
	if not o then return end
	for _, k in ipairs(BOXLIST) do if o[k] then pcall(function() o[k]:Remove() end) end end
	for _, l in ipairs(o.bones) do pcall(function() l:Remove() end) end
	if o.cham then o.cham:Destroy() end
	espCache[plr] = nil
end

local function addESP(plr)
	if plr == LP or espCache[plr] then return end

	for _, k in ipairs(BOXLIST) do if o[k] then o[k].Visible = false end end

local function bone(o, i, a, b, col)
	local l = o.bones[i]
	if not l then l = Draw.new("Line"); o.bones[i] = l end

	l.From = a; l.To = b; l.Color = col; l.Visible = true

local function renderESP()
	local vp = Camera.ViewportSize
	for plr, o in pairs(espCache) do
		if not plr.Parent then
			delESP(plr)
		else
			local char, hum, hrp = charOf(plr)
			local show, dist = false, 0
			if S.esp.on and not S.misc.streamproof and char then
				dist = (Camera.CFrame.Position - hrp.Position).Magnitude
				show = dist <= S.esp.max and not (S.esp.team and isTeam(plr))
			end
			if o.cham then
				o.cham.Enabled = show and S.esp.cham and not S.misc.streamproof
				o.cham.Adornee = o.cham.Enabled and char or nil
				o.cham.FillColor = COLORS[S.esp.col] or COLORS.Green
				o.cham.FillTransparency = S.esp.chamTrans
			end
			if not o.box then
			elseif not show then
				hideESP(o)
			else
				local head = char:FindFirstChild("Head") or hrp
				local tp, ton = Camera:WorldToViewportPoint(head.Position + Vector3.new(0,0.85,0))
				local bp, bon = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0,2.9,0))
				if not (ton and bon) then
					hideESP(o)
				else
					local h = math.abs(bp.Y - tp.Y)
					local w = h * 0.55
					local x, y = tp.X - w/2, tp.Y
					local mh = hum.MaxHealth > 0 and hum.MaxHealth or 100
					local frac = math.clamp(hum.Health / mh, 0, 1)
					local col = COLORS[S.esp.col] or COLORS.Green
					if S.esp.healthCol then
						if frac < 0.35 then col = COLORS.Red elseif frac < 0.7 then col = COLORS.Yellow end
					end
					o.box.Visible = S.esp.box
					o.box.Size = Vector2.new(w,h); o.box.Position = Vector2.new(x,y); o.box.Color = col
					o.name.Visible = S.esp.name
					o.name.Size = S.esp.textSize
					o.name.Text = plr.Name .. (S.esp.dist and ("  ["..math.floor(dist).."m]") or "")
					o.name.Position = Vector2.new(tp.X, y - S.esp.textSize - 4); o.name.Color = col
					o.hpbg.Visible = S.esp.hp; o.hp.Visible = S.esp.hp
					o.hpbg.Size = Vector2.new(3,h); o.hpbg.Position = Vector2.new(x-6,y)
					o.hp.Size = Vector2.new(3,h*frac); o.hp.Position = Vector2.new(x-6, y+h*(1-frac)); o.hp.Color = col
					o.tracer.Visible = S.esp.tracer
					o.tracer.From = Vector2.new(vp.X/2, vp.Y); o.tracer.To = Vector2.new(tp.X, bp.Y); o.tracer.Color = col
					o.dot.Visible = S.esp.dot
					o.dot.Radius = 3; o.dot.Position = Vector2.new(tp.X, tp.Y); o.dot.Color = col
					local used = 0
					if S.esp.skeleton then
						local set = char:FindFirstChild("UpperTorso") and R15 or R6
						for _, j in ipairs(set) do
							local a, b = char:FindFirstChild(j[1]), char:FindFirstChild(j[2])
							if a and b then
								local ap, aon = Camera:WorldToViewportPoint(a.Position)
								local bq, bon2 = Camera:WorldToViewportPoint(b.Position)
								if aon and bon2 then
									used += 1
									bone(o, used, Vector2.new(ap.X,ap.Y), Vector2.new(bq.X,bq.Y), col)
								end
							end
						end
					end
					for i = used+1, #o.bones do o.bones[i].Visible = false end
				end
			end
		end
	end
end

local function tpNearest()
	local _, _, hrp = charOf(LP)
	if not hrp then return end
	local best, bd = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP then
			local _, _, h = charOf(plr)
			if h then
				local d = (hrp.Position - h.Position).Magnitude
				if d < bd then bd, best = d, h end
			end
		end
	end
	if best then hrp.CFrame = best.CFrame * CFrame.new(0,0,4) end
end

	local _, _, hrp = charOf(LP)

	local hit = workspace:Raycast(ray.Origin, ray.Direction * 2000)
	if hit then hrp.CFrame = CFrame.new(hit.Position + Vector3.new(0,3,0)) end
end

			Size = UDim2.fromOffset(s, h),
			BackgroundColor3 = cfg.col,
			BackgroundTransparency = math.clamp(cfg.trans + math.random(0,35)/100, 0, 0.9),
			BorderSizePixel = 0, ZIndex = 40, Active = false,
			Position = UDim2.new(math.random(), 0, math.random(), 0),
		}, partHolder)
		new("UICorner", {CornerRadius = UDim.new(1,0)}, f)
		local rot = math.random(0,360)
		f.Rotation = rot
		particles[i] = {
			f=f, vy=math.random(cfg.vmin*10, cfg.vmax*10)/1000,
			vx=math.random(-8,8)/1000, sway=math.random()*6.28,
			rot=rot, rs=cfg.spin and math.random(-cfg.spin, cfg.spin) or 0,
			tw=cfg.twinkle, base=f.BackgroundTransparency,
		}
	end
end

local rebuildQueued = false

	if rebuildQueued then return end
	rebuildQueued = true
	task.delay(0.15, function()
		rebuildQueued = false
		buildParticles()
	end)
end

keep(RunService.Heartbeat:Connect(function(dt)
	if #particles == 0 then return end
	local now = os.clock()
	for _, p in ipairs(particles) do
		local pos = p.f.Position
		local y = pos.Y.Scale + p.vy * dt
		local x = pos.X.Scale + p.vx * dt + math.sin(now*1.4 + p.sway) * 0.0008
		if y > 1.06 then y = -0.06; x = math.random() end
		if x > 1.06 then x = -0.05 elseif x < -0.05 then x = 1.05 end
		p.f.Position = UDim2.new(x, 0, y, 0)
		if p.rs ~= 0 then
			p.rot = (p.rot + p.rs * dt) % 360
			p.f.Rotation = p.rot
		end
		if p.tw then
			p.f.BackgroundTransparency = math.clamp(p.base + math.sin(now*3 + p.sway)*0.22, 0, 0.95)
		end
	end
end))

local function updateMini()
	if not mini then return end
	if S.misc.streamproof then mini.Visible = false; return end
	mini.Visible = true
	local _, _, lhrp = charOf(LP)
	if not lhrp then return end
	local cam = Camera.CFrame
	local right, fwd = cam.RightVector, cam.LookVector
	local scale = S.esp.minimapSize / math.max(S.esp.minimapZoom, 1)
	local maxR = S.esp.minimapSize/2 - 6
	local i = 0
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP then
			local char, hum, hrp = charOf(plr)
			if char then
				local rel = hrp.Position - lhrp.Position
				local x = rel:Dot(right) * scale
				local z = rel:Dot(fwd) * scale
				if math.abs(x) <= maxR and math.abs(z) <= maxR then
					i += 1
					local d = miniDots[i]
					if not d then
						d = new("Frame", {Size=UDim2.fromOffset(7,7), AnchorPoint=Vector2.new(0.5,0.5),
							BorderSizePixel=0, ZIndex=46}, mini)
						new("UICorner", {CornerRadius=UDim.new(1,0)}, d)
						miniDots[i] = d
					end
					d.Visible = true
					d.Position = UDim2.new(0.5, x, 0.5, -z)
					local col = COLORS[S.esp.col] or COLORS.Green
					local mh = hum.MaxHealth > 0 and hum.MaxHealth or 100
					if hum.Health/mh < 0.5 then col = COLORS.Yellow end
					d.BackgroundColor3 = col
				end
			end
		end
	end
	for j = i+1, #miniDots do miniDots[j].Visible = false end
end

local function updateWarn()
	if not warnLabel then return end
	if not S.misc.aimWarn or S.misc.streamproof then warnLabel.Visible = false; return end
	local _, _, lhrp = charOf(LP)
	if not lhrp then warnLabel.Visible = false; return end
	local myPos = lhrp.Position
	local thr = math.cos(math.rad(S.misc.warnAngle))
	local n = 0
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP and not isTeam(plr) then
			local char, hum, hrp = charOf(plr)
			if char then
				local d = (hrp.Position - myPos).Magnitude
				if d <= S.misc.warnDist then
					local toMe = (myPos - hrp.Position).Unit
					if hrp.CFrame.LookVector:Dot(toMe) >= thr then n += 1 end
				end
			end
		end
	end
	if n > 0 then
		warnLabel.Visible = true
		warnLabel.Text = L("VISE PAR").." "..n.." "..L("JOUEUR")..(n > 1 and "S" or "")
		warnLabel.BackgroundTransparency = 0.05 + math.abs(math.sin(os.clock()*4))*0.35
	else
		warnLabel.Visible = false
	end
end

local function encode()
	local out = {}
	for k, tbl in pairs(S) do
		if type(tbl) == "table" then
			out[k] = {}
			for k2, v in pairs(tbl) do
				if typeof(v) == "EnumItem" then out[k][k2] = {__e = tostring(v)}
				elseif typeof(v) == "Color3" then out[k][k2] = {__c = {v.R, v.G, v.B}}
				else out[k][k2] = v end
			end
		else
			out[k] = tbl
		end
	end
	return HttpService:JSONEncode(out)
end
local function decode(str)
	local data = HttpService:JSONDecode(str)
	for k, tbl in pairs(data) do
		if S[k] ~= nil then
			if type(tbl) == "table" and type(S[k]) == "table" then
				for k2, v in pairs(tbl) do
					if type(v) == "table" and v.__e then
						local cat, nm = tostring(v.__e):match("Enum%.([^%.]+)%.(.+)$")
						if cat and nm and Enum[cat] then S[k][k2] = Enum[cat][nm] end
					elseif type(v) == "table" and v.__c then
						S[k][k2] = Color3.new(v.__c[1], v.__c[2], v.__c[3])
					else S[k][k2] = v end
				end
			else
				S[k] = tbl
			end
		end
	end
	for _, e in ipairs(registry) do
		local v = S
		for i = 1, #e.path do v = v[e.path[i]] end
		if v ~= nil then pcall(function() e.fn(v, false) end) end
	end
	setFly(S.mov.fly); applyWorld()
	applyMenuColor(S.theme.color); queueRebuild(); buildMini()
	applyLang()
end

		local a1 = math.floor(h / 65536)
		local a0 = h % 65536
		h = (((a1 * 31) % KEY_MOD) * 65536 + a0 * 31 + string.byte(s, i)) % KEY_MOD
	end
	return h
end
local function keySig(typ, exp, nonce)
	return string.upper(string.format("%08x", math.floor(keyHash(typ.."|"..exp.."|"..nonce.."|"..KEY_SECRET))))
end
local function parseKey(key)
	key = string.upper(string.gsub(tostring(key), "%s", ""))
	local p = {}
	for seg in string.gmatch(key, "[^-]+") do p[#p+1] = seg end
	if #p ~= 5 or p[1] ~= "VERSACE" then return nil, "format" end
	local typ, exp, nonce, sig = p[2], p[3], p[4], p[5]
	if keySig(typ, exp, nonce) ~= sig then return nil, "signature" end
	local e = tonumber(exp)
	if not e then return nil, "format" end
	if typ ~= "LT" and e < os.time() then return nil, "expiree" end
	return {typ = typ, exp = e}
end
